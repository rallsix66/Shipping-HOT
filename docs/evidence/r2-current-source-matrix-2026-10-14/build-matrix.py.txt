def read(path): return json.loads(Path(path).read_text(encoding='utf-8'))
def ref(path, selector=None):
    p = Path(path)
    assert p.is_file(), str(p)
    return {'path':p.as_posix(), 'selector':selector, 'sha256':hashlib.sha256(p.read_bytes().replace(b'\r\n', b'\n')).hexdigest(), 'hashBasis':'committed_text_LF'}
def outcome(path, sid):
    x = read(path)
    return next(r for r in x['results'] if r['id'] == sid)
def fetch(path, sid):
    x = outcome(path, sid)
    return {'evidence':ref(path, {'id':sid}), 'url':x['url'],
        'startedAt':x.get('startedAt',x.get('started')), 'receivedAt':x.get('receivedAt'),
        'finishedAt':x.get('finishedAt',x.get('finished')), 'httpStatus':x.get('httpStatus'),
        'wireSha256':x.get('fetchedBytesSha256',x.get('rawResponseSha256',x.get('wireSha256'))),
        'bytes':x.get('rawByteLength',x.get('bytes')), 'scope':'one_actual_index_response',
        'verbatimLatestRawRetained':x.get('rawLocalOnly',x.get('rawStructureFixtureRetained',False)),
        'notContinuousProduction':True}

from pathlib import Path
from datetime import datetime, timezone
import json, re

base = Path('C:/Users/Administrator/AppData/Roaming/io.github.clash-verge-rev.clash-verge-rev')
out = Path('docs/evidence/r2-carrier-reapplied-2026-10-15/applied-config-check.json')
hosts = ['www.maersk.com', 'www.one-line.com']
files = []

def stat_record(p):
    s = p.stat()
    return (s.st_size, s.st_mtime_ns)

index = base / 'profiles.yaml'
before = stat_record(index)
items = []; current = None; item = None
# Only allowlisted linkage keys are retained; URLs/credentials/proxy fields are discarded.
with index.open(encoding='utf-8-sig') as stream:
    for line in stream:
        match = re.match(r'^current:\s*["\x27]?([A-Za-z0-9_-]+)', line)
        if match: current = match.group(1)
        match = re.match(r'^\s*- uid:\s*(\S+)', line)
        if match:
            item = {'uid': match.group(1)}; items.append(item); continue
        match = re.match(r'^\s+(type|file|script|merge):\s*["\x27]?([A-Za-z0-9_.-]+)', line)
        if match and item is not None: item[match.group(1)] = match.group(2)
files.append({'role': 'profile_linkage_only', 'unchangedSizeAndMtime': before == stat_record(index)})
selected = next(i for i in items if i['uid'] == current)
script = next(i for i in items if i['uid'] == selected['script'])

def extension_presence(name, role):
    p = base / 'profiles' / name
    before = stat_record(p)
    present = dict.fromkeys(hosts, False)
    with p.open(encoding='utf-8-sig') as stream:
        for line in stream:
            for host in hosts:
                if re.search(r'["\x27]' + re.escape(host) + r'["\x27]', line): present[host] = True
    rec = {'role': role, 'file': 'profiles/' + name, 'exactHostLiteralPresent': present,
           'unchangedSizeAndMtime': before == stat_record(p), 'executedOrEvaluated': False}
    files.append(rec)
    return rec

selected_script = extension_presence(script['file'], 'selected_subscription_extension')
global_script = extension_presence('Script.js', 'global_extension')

def dns_presence(name, role):
    p = base / name; before = stat_record(p)
    in_dns = False; in_filter = False; filters = []; modes = {}
    with p.open(encoding='utf-8-sig') as stream:
        for line in stream:
            if not in_dns:
                if re.match(r'^dns:\s*(?:#.*)?$', line): in_dns = True
                continue
            if re.match(r'^\S', line): break
            match = re.match(r'^\s+(enhanced-mode|fake-ip-filter-mode):\s*([A-Za-z-]+)', line)
            if match: modes[match.group(1)] = match.group(2)
            if re.match(r'^\s+fake-ip-filter:\s*(?:#.*)?$', line): in_filter = True; continue
            if in_filter:
                match = re.match(r'^\s*-\s*["\x27]?([^\s"\x27#]+)["\x27]?\s*(?:#.*)?$', line)
                if match: filters.append(match.group(1))
                elif re.match(r'^\s+[A-Za-z][A-Za-z0-9_-]*:', line): in_filter = False
    rec = {'role': role, 'file': name, 'dnsBlockFound': in_dns, 'modes': modes,
           'filterValues': filters, 'filterCount': len(filters), 'exactHostInFakeIpFilter': {h: h in filters for h in hosts},
           'unchangedSizeAndMtime': before == stat_record(p)}
    files.append(rec)
    return rec

generated = dns_presence('clash-verge.yaml', 'generated_config_on_disk')
base_config = dns_presence('config.yaml', 'application_base_config_on_disk')
result = {'checkedAt': datetime.now(timezone.utc).isoformat(), 'readOnly': True,

          'credentialFieldsInspectedOrExported': False,
          'scope': 'Allowlisted profile linkage, exact quoted target-host literals and DNS keys only; no full config, subscription URL, proxy, secret or credential values exported. Disk presence is not a running-core/API verification.',
          'files': files}
previous=json.loads(Path('.tmp/clash-two-hosts-preview-latest.json').read_text(encoding='utf-8'))
old_text=Path(previous['backup']).read_text(encoding='utf-8-sig')
old_hosts=re.findall(r'^\s+"([^"\r\n]+)",\s*$',old_text,re.M)
new_text=(base/'profiles'/script['file']).read_text(encoding='utf-8-sig')
new_hosts=re.findall(r'^\s+"([^"\r\n]+)",\s*$',new_text,re.M)
result['originalFilterCount']=len(old_hosts)
result['selectedExtensionCount']=len(new_hosts)
result['selectedExtensionExactlyOriginal21Plus2']=new_hosts==old_hosts+hosts and len(old_hosts)==21
result['original21Preserved']=all(h in new_hosts for h in old_hosts)
result['generatedExactly23TargetsPresent']=generated['filterCount']==23 and all(generated['exactHostInFakeIpFilter'].values())
result['generatedExactlyOriginal21Plus2']=generated.pop('filterValues')==old_hosts+hosts
base_config.pop('filterValues')
result['userAppliedChange']=True
result['agentSystemWriteExecuted']=False
out.parent.mkdir(exist_ok=True)

assert result['selectedExtensionExactlyOriginal21Plus2'] and result['generatedExactly23TargetsPresent'] and result['original21Preserved'] and result['generatedExactlyOriginal21Plus2'], 'Applied configuration mismatch; do not start DNS/HTTP'
print(json.dumps(result, ensure_ascii=True))

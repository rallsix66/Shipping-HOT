def run(args,label):

    r=subprocess.run(args,capture_output=True)

        except UnicodeDecodeError:return b.decode('gb18030')
    rec={'label':label,'command':args,'startedAt':started,'finishedAt':datetime.now(timezone.utc).isoformat(),'stdout':decode(r.stdout),'stderr':decode(r.stderr),'rawExit':r.returncode}
    commands.append(rec);print(json.dumps({'label':label,'rawExit':r.returncode,'stdoutTail':rec['stdout'][-1200:],'stderrTail':rec['stderr'][-800:]},ensure_ascii=True));return r

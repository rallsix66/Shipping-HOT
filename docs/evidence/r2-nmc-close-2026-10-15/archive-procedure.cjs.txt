const fs = require('node:fs'), cp = require('node:child_process'), crypto = require('node:crypto');
const d = 'docs/evidence/r2-nmc-close-2026-10-15', base = 'f58e01e7360f17ab55706e36ce09851e265ddedb', code = 'bebe1441465adfcabb9daf93cb2a8656ec1fc796';
const read = p => fs.readFileSync(p, 'utf8'), sha = s => crypto.createHash('sha256').update(s.replaceAll('\r\n', '\n')).digest('hex'), git = (...a) => cp.execFileSync('git', a, { encoding: 'utf8' });
if (JSON.parse(read(d + '/validation.json')).rawExit !== 0 || JSON.parse(read(d + '/doc-lint.json')).rawExit !== 0) throw Error('checks_pending');
const productDiff = git('diff', code, '--', 'server', 'src', 'shared', 'scripts', 'package.json', 'pnpm-lock.yaml', 'nitro.config.ts', '.github'), catalogueDiff = git('diff', base, '--', 'docs/intel-source-catalog.md');
if (productDiff || catalogueDiff) throw Error('unexpected_product_or_catalogue_change');
fs.writeFileSync(d + '/source-proof.json', JSON.stringify({ executionHead: base, productCodeSha: code, productDiff, catalogueDiff, sourceRequests: 0, systemDns: 0, scope: 'Only approved NMC historical evidence mapping/qualification and bounded current-document corrections; no product/schema/production change', productGate: 'NOT_RUN_no_product_change' }, null, 2) + '\n');
const files = [...new Set([...git('diff', '--name-only', base).trim().split('\n').filter(Boolean), ...fs.readdirSync(d).filter(n => n !== 'archive-manifest.json').map(n => d + '/' + n)])].sort();
if (files.some(p => !p.startsWith('docs/'))) throw Error('non_docs_change');
fs.writeFileSync(d + '/archive-manifest.json', JSON.stringify({ baselineHead: base, productCodeSha: code, files: files.map(p => ({ path: p, committedTextLfSha256: sha(read(p)) })), sourceRequests: 0, systemDns: 0, productGate: 'NOT_RUN_no_product_change' }, null, 2) + '\n');
console.log(JSON.stringify({ files: files.length, productDiff, catalogueDiff, sourceRequests: 0 }));

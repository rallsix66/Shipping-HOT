const sensitive = /authorization|bearer|cookie|api[_-]?key|password|credential|token|secret|csrf/i
function sanitizeLocation(value: string | string[] | undefined) {
  if (value === undefined) return { present: false }
  if (Array.isArray(value) || value.length > 4096 || /[\r\n]/.test(value)) return { present: true, sanitized: null, redacted: true, reason: "invalid_or_oversized" }
  try {
    const literal = value.trim(), u = new URL(literal, "https://synthetic-location-base.invalid/")
    if (!["http:", "https:"].includes(u.protocol)) return { present: true, sanitized: null, redacted: true, reason: "non_http" }
    const relative = !/^[A-Za-z][A-Za-z0-9+.-]*:/.test(literal)
    let pathname = relative && !literal.startsWith("/") ? literal.split(/[?#]/, 1)[0] : u.pathname
    const pathRedacted = pathname.split("/").some(p => p.length >= 32 || sensitive.test(p))
    pathname = pathname.split("/").map(p => p.length >= 32 || sensitive.test(p) ? "[path-redacted]" : p).join("/")
    const sanitized = relative ? literal.startsWith("//") ? `//${u.host}${pathname}` : pathname : `${u.origin}${pathname}`
    return { present: true, sanitized, relative, credentialsRemoved: Boolean(u.username || u.password), queryRemoved: Boolean(u.search), fragmentRemoved: Boolean(u.hash), pathRedacted, followed: false }
  } catch { return { present: true, sanitized: null, redacted: true, reason: "invalid_location" } }
}

function redactErrorMessage(value: unknown) {
  const raw = typeof value === "string" ? value : "unavailable_error_message"
  let message = raw.replace(/https?:\/\/[^\s<>"']+/gi, u => sanitizeLocation(u)?.sanitized ?? "[url-redacted]")
  message = message.replace(/\b(authorization|proxy-authorization|cookie|set-cookie)\b\s*[:=][^\r\n]*/gi, "$1=[redacted]")
    .replace(/\b(api[_-]?key|access[_-]?token|token|secret|password|credential|csrf)\s*[:=]\s*(?:"[^"]*"|'[^']*'|[^,\s;]+)/gi, "$1=[redacted]")
    .replace(/\b(?:Bearer|Basic)\s+[A-Za-z0-9._~+\/-]+=*/gi, "[authorization-redacted]")
    .replace(/\bsk-[A-Za-z0-9_-]{12,}\b/g, "[key-redacted]")
    .replace(/[\u0000-\u001f\u007f]+/g, " ").trim()
  return { message: message.slice(0, 512), redacted: message !== raw, truncated: message.length > 512 }
}
function describeError(error: unknown) {
  const code = (error as any)?.code
  return { code: typeof code === "string" && /^(?:E[A-Z0-9_]{2,45}|ERR_[A-Z0-9_]{2,45})$/.test(code) ? code : null,
    ...redactErrorMessage(error instanceof Error ? error.message : "unknown_transport_error"), stackSaved: false }
}
async function validateSyntheticFailureRecording() {
  const marker = "SYNTHETIC_ONLY_VALUE"
  const original = new Error(`synthetic connection reset; token=${marker} https://synthetic-user:synthetic-pass@synthetic.example.test/news?api_key=${marker}#fragment\nCookie: ${marker}`) as Error & { code: string }
  original.code = "ECONNRESET"
  let fakeTransportCalls = 0
  const captured = describeError(original)
  const failure = await secureFetchArticle("https://synthetic.example.test/news", {
    policy: { sourceId: "synthetic-error-only", status: "allowed", fetchAllowed: true, persistence: "disallowed", allowedHosts: ["synthetic.example.test"], allowedContentTypes: ["text/html"] },
    lookup: async () => [{ address: "93.184.216.34", family: 4 }],
    transport: async () => { fakeTransportCalls++; throw original }, maxRedirects: 0,
  })
  expect(failure.ok).toBe(false)
  if (failure.ok) throw new Error("synthetic_failure_expected")
  const secure = { code: failure.code, ...redactErrorMessage(failure.message) }
  expect(secure.code).toBe("fetch_failed"); expect(captured.code).toBe("ECONNRESET")
  for (const result of [secure, captured]) {
    expect(result.message).toContain("synthetic connection reset")
    for (const secret of [marker, "synthetic-user", "synthetic-pass", "api_key=", "#fragment"]) expect(result.message).not.toContain(secret)
  }
  expect(fakeTransportCalls).toBe(1)
  save("synthetic-error-recording", { source: "synthetic_only_not_Maersk", injectedLookup: true, fakeTransportCalls, sourceDnsCalls: 0, realTransportCalls: 0, MaerskRequests: 0, secureEntryError: secure, transportError: captured, redactionAssertions: "PASS", role: "Offline error recording proof, not diagnosis of any real provider" })
}

function safeUrl(s: string, base: string) {
  try { const u = new URL(s, base); return ["http:", "https:"].includes(u.protocol) && !u.username && !u.password && ![...u.searchParams.keys()].some(k => sensitive.test(k) || /signature|^sig$|utm_|gclid|fbclid/i.test(k)) ? u.href : null }
  catch { return null }
}
function observe(body: string, url: string) {
  const $ = load(body); $("script,style,noscript,iframe,input,meta,link").remove()
  const candidates: unknown[] = []
  const date = /\b\d{1,4}[/-]\d{1,2}[/-]\d{1,4}\b|\d{4}年\d{1,2}月\d{1,2}日/g
  $("a[href]").each((index, el) => {
    if (candidates.length >= 8) return
    const a = $(el), href = safeUrl(a.attr("href") ?? "", url), text = a.text().replace(/\s+/g, " ").trim()
    if (!href || !text || sensitive.test(text)) return
    let container = a.parent(), dates: string[] = [], depth = 0
    while (depth++ < 3) { dates = [...new Set(container.text().match(date) ?? [])]; if (dates.length) break; container = container.parent() }
    if (!dates.length) return
    candidates.push({ index, titleExcerpt: text.slice(0, 100), href, dateCandidates: dates.slice(0, 5), ancestorDepth: depth, localAnchorCount: container.find("a[href]").length, association: "observation_only_not_assigned" })
  })
  return { kind: "sanitized bounded structural metadata, not raw HTML or configured fields", counts: { anchors: $("a[href]").length, tables: $("table").length, times: $("time").length }, candidates, parser: "NOT_RUN", fullFieldsPass: false }
}

import { it, expect } from "vitest"
import { lookup } from "node:dns/promises"
import { existsSync, mkdirSync, readFileSync, writeFileSync } from "node:fs"
import { createHash } from "node:crypto"
import { execFileSync } from "node:child_process"
import { secureFetchArticle, defaultTransport, isBlockedAddress } from "../server/services/article-fetch-security"
const d = "docs/evidence/r2-id-t02-location-2026-10-15", local = ".tmp/r2-id-t02-location-20261015-private", approvedHost = "jdih.kemenkeu.go.id"
const hash = (v: string | Uint8Array) => createHash("sha256").update(v).digest("hex"), save = (n: string, v: unknown) => writeFileSync(`${d}/${n}.json`, JSON.stringify(v, null, 2) + "\n")
const safeError = (e: any) => ({ code: typeof e?.code === "string" ? e.code : null, message: String(e?.message ?? "unknown_error").replace(/https?:\/\/\S+/g, "[url-redacted]").replace(/\b(cookie|authorization|token|secret|password|api[_-]?key)\s*[:=].*/gi, "[redacted]").replace(/[\r\n]/g, " ").slice(0, 350) })
const endpoint = (u: URL) => ({ protocol: u.protocol, hostname: u.hostname, port: u.port || null, pathname: u.pathname.replace(/[A-Za-z0-9_-]{32,}/g, "[long-segment-redacted]"), queryPresent: !!u.search, querySha256: u.search ? hash(u.search) : null, credentialsPresent: !!(u.username || u.password), fragmentPresent: !!u.hash })
it("one exact ID-T02 catalogue GET and at most one independently validated real Location target", async () => {
  expect(process.env.SHIPPING_APPROVED_ID_T02_LOCATION).toBe("1"); mkdirSync(d, { recursive: true }); mkdirSync(local, { recursive: true }); if (existsSync(`${d}/manifest.json`)) throw Error("already_reserved_never_repeat")
  const url = "https://jdih.kemenkeu.go.id/", head = execFileSync("git", ["rev-parse", "HEAD"], { encoding: "utf8" }).trim(), row = readFileSync("docs/intel-source-catalog.md", "utf8").split(/\r?\n/).find(l => l.startsWith("| ID-T02 |"))!
  expect(head).toBe("4ca3a77cd8cdb1be3c793052b7edff628cec0529"); expect(row.split("|")[4].trim()).toBe(url)
  const matrix = JSON.parse(readFileSync("docs/evidence/r2-current-source-matrix-2026-10-14/mapping.json", "utf8")); expect(matrix.entries.find((s: any) => s.id === "XX-K03").r2_4.verdict).toBe("OK"); expect(matrix.measuredCoverage.r2_5.numerator).toBe(22)
  save("manifest", { executionHead: head, productCodeSha: "0a483dd18a2027c73bf950cc0b948c013e29cf09", id: "ID-T02", exactCatalogueUrl: url, catalogueRowLfSha256: hash(row), allowedHost: approvedHost, maximumSystemDnsCalls: 2, maximumGetAttempts: 2, additionalGetCondition: "Actual Location only; same approved exact host/HTTPS/no credentials/no abnormal port, then existing safety boundary; never a second redirect", retries: 0, automaticRedirects: 0, details: 0, pagination: 0, rawLocalOnly: true, persistence: "disallowed", systemSettingsWritten: false, productChanged: false })
  const dns: any[] = [], requests: any[] = [], results: any[] = []; save("dns-ledger", dns); save("request-ledger", requests)
  async function one(target: string, role: string) {
    const u = new URL(target); expect(u.hostname).toBe(approvedHost); expect(u.protocol).toBe("https:"); expect(u.username + u.password + u.port).toBe("")
    const result: any = { role, endpoint: endpoint(u), startedAt: new Date().toISOString(), systemDnsCalls: 0, getAttempts: 0, status: "NOT_RUN" }; results.push(result)
    let location: string | null = null
    try {
      if (dns.length >= 2 || requests.length >= 2) throw Error("budget_exhausted"); result.systemDnsCalls = 1; const n: any = { role, hostname: u.hostname, reservedAt: new Date().toISOString(), state: "reserved_unknown" }; dns.push(n); save("dns-ledger", dns)
      let timer: ReturnType<typeof setTimeout> | undefined, addresses: any[]
      try { addresses = await Promise.race([lookup(u.hostname, { all: true, verbatim: true }), new Promise<never>((_, reject) => { timer = setTimeout(() => reject(Error("system_dns_timeout")), 10000) })]); Object.assign(n, { addresses, classification: !addresses.length ? "dns_empty" : addresses.some(isBlockedAddress) ? "blocked_address" : "public_address", state: "finished" }); result.addressClassification = n.classification }
      catch (e) { Object.assign(n, { state: "failed", error: safeError(e) }); throw e }
      finally { if (timer) clearTimeout(timer); n.finishedAt = new Date().toISOString(); save("dns-ledger", dns) }
      const r = await secureFetchArticle(target, { policy: { sourceId: "ID-T02", status: "allowed", fetchAllowed: true, persistence: "disallowed", allowedHosts: [approvedHost], allowedContentTypes: ["text/html", "application/xhtml+xml"] }, lookup: async host => { expect(host).toBe(approvedHost); return addresses }, maxBytes: 1048576, timeoutMs: 10000, maxRedirects: 0, accept: "text/html, application/xhtml+xml", transport: async (t, ips, options) => {
        if (result.getAttempts || requests.length >= 2 || t.href !== target || options.caPem) throw Error("one_shot_boundary"); result.getAttempts++
        const q: any = { role, method: "GET", endpoint: endpoint(t), reservedAt: new Date().toISOString(), state: "reserved_unknown" }; requests.push(q); save("request-ledger", requests)
        let response; try { response = await defaultTransport(t, ips, options) } catch (e) { result.transportError = safeError(e); throw e }
        const raw = typeof response.body === "string" ? Buffer.from(response.body) : response.body; writeFileSync(`${local}/${role}.response.bin`, raw, { flag: "wx" })
        const loc = Array.isArray(response.headers.location) ? response.headers.location[0] : response.headers.location; location = loc ?? null
        let locMeta: any = null
        if (location !== null) { writeFileSync(`${local}/${role}.location.json`, JSON.stringify({ rawLocation: location, requestedUrl: target }, null, 2), { flag: "wx" }); try { locMeta = { rawHeaderSha256: hash(location), rawValueLength: location.length, resolved: endpoint(new URL(location, target)), followedAutomatically: false } } catch { locMeta = { invalid: true, rawHeaderSha256: hash(location), rawValueLength: location.length, followedAutomatically: false } } }
        Object.assign(result, { receivedAt: new Date().toISOString(), httpStatus: response.status, contentType: response.headers["content-type"] ?? null, bytes: raw.length, wireSha256: hash(raw), completeResponseCaptured: true, location: locMeta }); return response
      } })
      if (r.ok) { result.status = "CAPTURED_SAFE_HTML"; result.decoding = r.decoding } else { result.status = "BLOCKED"; result.secureEntryError = { code: r.code, message: safeError({ message: r.message }).message }; result.httpStatus ??= r.status ?? null }
    } catch (e) { result.status = "BLOCKED"; result.error = safeError(e) }
    finally { result.finishedAt = new Date().toISOString(); const q = requests.find(q => q.role === role); if (q) { q.state = result.completeResponseCaptured ? "complete_response_captured" : "finished_without_complete_response"; q.finishedAt = result.finishedAt; save("request-ledger", requests) } save(role + "-capture", result) }
    return { result, location }
  }
  const root = await one(url, "catalogue"), decision: any = { actualLocationPresent: root.location !== null, extraTargetGetAllowed: false, extraTargetGetAttempted: false, actualLocationOnly: true }
  if (root.result.completeResponseCaptured && [301, 302, 303, 307, 308].includes(root.result.httpStatus) && root.location !== null) {
    try {
      const target = new URL(root.location, url), sensitiveQuery = [...target.searchParams.keys()].some(k => /token|password|credential|secret|signature|auth|session|api.?key/i.test(k))
      const checks = { noControlCharacters: !/[\r\n\u0000-\u001f]/.test(root.location), sameApprovedHost: target.hostname === approvedHost, https: target.protocol === "https:", noCredentials: !(target.username || target.password) && !sensitiveQuery, noAbnormalPort: !target.port, noFragment: !target.hash, notSelfRedirect: target.href !== url }; decision.checks = checks; decision.resolved = endpoint(target)
      if (Object.values(checks).every(Boolean)) { decision.extraTargetGetAllowed = true; decision.extraTargetGetAttempted = true; await one(target.href, "location-target"); decision.secondRedirectFollowed = false } else decision.reason = "Location outside bounded target conditions; no target DNS orGET"
    } catch (e) { decision.reason = "Location parse failed; no target DNS orGET"; decision.error = safeError(e) }
  } else decision.reason = root.result.addressClassification === "blocked_address" ? "Existing DNS safety boundary blocked root; stop this source" : "No complete redirect with actual Location; no conditional target request"
  save("location-decision", decision)
  save("capture-summary", { results, systemDnsCalls: dns.length, getAttempts: requests.length, completeResponses: results.filter(r => r.completeResponseCaptured).length, conditionalTargetAttempted: decision.extraTargetGetAttempted, automaticRedirects: 0, retries: 0, details: 0, pagination: 0, otherSourceRequests: 0, rawRedistributed: false, credentialsUsed: false, systemSettingsWritten: false, productChanged: false, recordsPersisted: 0 })
  expect(dns.length).toBeLessThanOrEqual(2); expect(requests.length).toBeLessThanOrEqual(2); expect(results.length).toBeLessThanOrEqual(2)
}, 50000)

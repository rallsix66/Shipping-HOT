import { it, expect } from "vitest"
import { lookup } from "node:dns/promises"
import { existsSync, mkdirSync, readFileSync, writeFileSync } from "node:fs"
import { createHash } from "node:crypto"
import { execFileSync } from "node:child_process"
import { secureFetchArticle, defaultTransport, isBlockedAddress } from "../server/services/article-fetch-security"

const dir = "docs/evidence/r2-official-gaps-2026-10-15"
const local = ".tmp/r2-official-gaps-20261015-private"
const hash = (v: string | Uint8Array) => createHash("sha256").update(v).digest("hex")
const save = (n: string, v: unknown) => writeFileSync(`${dir}/${n}.json`, JSON.stringify(v, null, 2) + "\n")
const read = (n: string) => JSON.parse(readFileSync(`${dir}/${n}.json`, "utf8"))
function errorMetadata(error: unknown) {
  const message = error instanceof Error ? error.message : "unknown_error"
  return { code: typeof (error as any)?.code === "string" ? (error as any).code : null, message: message.replace(/https?:\/\/\S+/g, "[url-redacted]").replace(/\b(cookie|authorization|token|secret|password|api[_-]?key)\s*[:=].*/gi, "[redacted]").replace(/[\r\n]/g, " ").slice(0, 400) }
}
function locationMetadata(value: string | string[] | undefined, url: string) {
  const location = Array.isArray(value) ? value[0] : value
  if (!location) return null
  try { const u = new URL(location, url); return { targetOriginAndPath: u.origin + u.pathname, queryPresent: !!u.search, fragmentPresent: !!u.hash, followed: false } }
  catch { return { invalid: true, valueSha256: hash(location), followed: false } }
}
it("capture only delegated official gap targets once, with safe DNS and no redirects/retries", async () => {
  expect(process.env.SHIPPING_APPROVED_OFFICIAL_GAPS).toBe("1")
  const phase = process.env.SHIPPING_OFFICIAL_PHASE ?? "initial"
  const fixed = [
    { id: "PH-W01", url: "https://www.pagasa.dost.gov.ph/tropical-cyclone/severe-weather-bulletin" },
    { id: "PH-W02", url: "https://www.pagasa.dost.gov.ph/weather/weather-advisory" },
    { id: "ID-W02-root", url: "https://maritim.bmkg.go.id/marine2026-data/" },
    { id: "XX-W02-targetTc", url: "https://www.jma.go.jp/bosai/typhoon/data/targetTc.json" },
  ]
  mkdirSync(dir, { recursive: true }); mkdirSync(local, { recursive: true })
  let dns: any[] = [], requests: any[] = [], results: any[] = [], targets: { id: string, url: string }[]
  if (phase === "initial") {
    if (existsSync(`${dir}/manifest.json`)) throw Error("initial_already_reserved_never_repeat")
    const catalog = readFileSync("docs/intel-source-catalog.md", "utf8")
    for (const row of fixed.slice(0, 3)) expect(catalog).toContain(row.url)
    const executionHead = execFileSync("git", ["rev-parse", "HEAD"], { encoding: "utf8" }).trim()
    expect(executionHead).toBe("cfce1b3f0a1cccf8cc61527f5aebea879a8a1aec")
    save("manifest", { executionHead, productCodeSha: "bebe1441465adfcabb9daf93cb2a8656ec1fc796", authorizedAt: "2026-10-15 delegated instruction after accepted NMC closeout", fixed, maximumGetAttempts: 7, conditional: "one same-host apidoc explicitly linked by BMKG root; one actual targetTc object forecast+specifications pair; no guessed IDs/URLs", sourceRequestsOnly: true, sourcePolicy: { fetchAllowed: true, persistence: "disallowed", rawLocalOnly: true, remote: "minimal derived metadata and hashes", cookiesOrKeysRecorded: false }, maxBytes: 1048576, perTargetTimeoutMs: 10000, retries: 0, redirectsFollowed: 0, pdfDownloads: 0, portJsonRequests: 0, providerOrProductionChanged: false })
    targets = fixed
  } else {
    if (phase !== "conditional") throw Error("invalid_phase")
    dns = read("dns-ledger"); requests = read("request-ledger"); results = read("capture-summary").results
    const plan = read("conditional-targets")
    expect(plan.manuallyReviewedAgainstInitialResponse).toBe(true)
    targets = plan.targets
    expect(targets.length).toBeLessThanOrEqual(3)
    for (const row of targets) {
      const u = new URL(row.url)
      if (row.id === "ID-W02-apidoc") {
        expect(u.hostname).toBe("maritim.bmkg.go.id")
        expect(plan.bmkgObservedLinks).toContain(row.url)
      } else {
        expect(["XX-W02-forecast", "XX-W02-specifications"]).toContain(row.id)
        expect(u.hostname).toBe("www.jma.go.jp")
        expect(plan.actualTropicalCyclone).toMatch(/^TC[0-9]+$/)
        expect(row.url).toBe(`https://www.jma.go.jp/bosai/typhoon/data/${plan.actualTropicalCyclone}/${row.id === "XX-W02-forecast" ? "forecast" : "specifications"}.json`)
      }
    }
  }
  save("dns-ledger", dns); save("request-ledger", requests)
  const batch = new AbortController(), timer = setTimeout(() => batch.abort(), 55000)
  try {
    for (const row of targets!) {
      if (dns.some(d => d.id === row.id) || results.some(r => r.id === row.id)) throw Error("target_already_reserved_never_repeat")
      const u = new URL(row.url)
      expect(u.protocol).toBe("https:"); expect(u.username + u.password + u.search + u.hash).toBe("")
      const result: any = { ...row, phase, host: u.hostname, startedAt: new Date().toISOString(), systemDnsCalls: 0, getAttempts: 0, status: "NOT_RUN", rawLocalOnly: true, qualificationChanged: false }
      try {
        if (batch.signal.aborted || dns.length >= 7) throw Error("batch_budget_exhausted")
        result.systemDnsCalls = 1
        const record: any = { id: row.id, host: u.hostname, url: row.url, reservedAt: new Date().toISOString(), state: "reserved_unknown" }; dns.push(record); save("dns-ledger", dns)
        let dnsTimer: ReturnType<typeof setTimeout> | undefined, addresses: any[]
        try {
          addresses = await Promise.race([lookup(u.hostname, { all: true, verbatim: true }), new Promise<never>((_, reject) => { dnsTimer = setTimeout(() => reject(Error("system_dns_timeout")), 10000) })])
          Object.assign(record, { addresses, classification: !addresses.length ? "dns_empty" : addresses.some(isBlockedAddress) ? "blocked_address" : "public_address", state: "finished" })
        } catch (error) { Object.assign(record, { state: "failed", error: errorMetadata(error) }); throw error }
        finally { if (dnsTimer) clearTimeout(dnsTimer); record.finishedAt = new Date().toISOString(); save("dns-ledger", dns) }
        result.addressClassification = record.classification
        const fetched = await secureFetchArticle(row.url, {
          policy: { sourceId: row.id, status: "allowed", fetchAllowed: true, persistence: "disallowed", allowedHosts: [u.hostname], allowedContentTypes: ["text/html", "application/xhtml+xml", "application/json", "text/json", "text/plain"] },
          maxBytes: 1048576, timeoutMs: 10000, maxRedirects: 0, signal: batch.signal, accept: "text/html, application/json, text/plain",
          lookup: async host => { expect(host).toBe(u.hostname); return addresses },
          transport: async (url, ips, options) => {
            if (result.getAttempts || requests.length >= 7 || url.href !== row.url || options.caPem) throw Error("one_shot_boundary")
            result.getAttempts++
            const req: any = { id: row.id, method: "GET", url: row.url, reservedAt: new Date().toISOString(), state: "reserved_unknown" }; requests.push(req); save("request-ledger", requests)
            let response
            try { response = await defaultTransport(url, ips, options) }
            catch (error) { result.transportError = errorMetadata(error); throw error }
            const raw = typeof response.body === "string" ? Buffer.from(response.body) : response.body
            writeFileSync(`${local}/${row.id}.response.bin`, raw, { flag: "wx" })
            Object.assign(result, { receivedAt: new Date().toISOString(), httpStatus: response.status, contentType: response.headers["content-type"] ?? null, bytes: raw.length, wireSha256: hash(raw), completeResponseCaptured: true, location: locationMetadata(response.headers.location, row.url) })
            return response
          },
        })
        if (fetched.ok) { result.status = "CAPTURED_SAFE_RESPONSE"; result.decoding = fetched.decoding }
        else { result.status = "BLOCKED"; result.secureEntryError = { code: fetched.code, message: fetched.message }; result.httpStatus ??= fetched.status ?? null }
      } catch (error) { result.status = "BLOCKED"; result.error = errorMetadata(error) }
      finally {
        result.finishedAt = new Date().toISOString(); results.push(result); save(row.id + "-capture", result)
        const req = requests.find(r => r.id === row.id)
        if (req) { req.state = result.completeResponseCaptured ? "complete_response_captured" : "finished_without_complete_response"; req.finishedAt = result.finishedAt; save("request-ledger", requests) }
        save("capture-summary", { results, systemDnsCalls: dns.length, getAttempts: requests.length, completeResponses: results.filter(r => r.completeResponseCaptured).length, safeSuccessfulResponses: results.filter(r => r.status === "CAPTURED_SAFE_RESPONSE").length, retries: 0, redirectsFollowed: 0, pdfDownloads: 0, portJsonRequests: 0, rawRedistributed: false, credentialsUsed: false, networkSettingsChanged: false, qualificationChanged: false, overallR2Pass: false })
      }
    }
  } finally { clearTimeout(timer) }
  expect(requests.length).toBeLessThanOrEqual(7); expect(dns.length).toBeLessThanOrEqual(7)
}, 60000)

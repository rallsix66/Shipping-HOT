const fs=require('node:fs'),cp=require('node:child_process')
const dir='docs/evidence/r2-one-newsroom-replay-final-2026-10-15';fs.mkdirSync(dir,{recursive:true})
if(fs.existsSync(dir+'/launch.json'))throw Error('replay_already_launched')
const sha=cp.execFileSync('git',['rev-parse','HEAD'],{encoding:'utf8'}).trim()
if(sha!==fs.readFileSync('.tmp/one-newsroom-code-sha.txt','utf8').trim())throw Error('wrong_code_sha')
const args=['node_modules/vitest/vitest.mjs','run','-c','.tmp/one-newsroom-raw-replay-final-20261015.config.ts'],startedAt=new Date().toISOString()

const r=cp.spawnSync(process.execPath,args,{windowsHide:true,encoding:'utf8',maxBuffer:1<<24})
